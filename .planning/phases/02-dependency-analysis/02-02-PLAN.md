    ```typescript
    // --- Dependency analysis (Phase 2) ---
    const prdPath = initialState.sourceFile;
    if (prdPath && fs.existsSync(prdPath)) {
      const rawMarkdown = fs.readFileSync(prdPath, 'utf-8');
      const parsedPrd = parsePrd(rawMarkdown);
      const deps = inferDependencies(parsedPrd.stories);
      const graph = buildExecutionGraph(parsedPrd.stories, deps);
      const report = formatDependencyReport(parsedPrd.stories, graph, deps);
      console.log(report);

      if (graph.cycles.length > 0) {
        process.exit(1);
      }
    }
    ```

    Key implementation notes:
    - `parsePrd` is already imported in workspace.ts (used in createWorkspace). No new import needed for it.
